"use client";

import { useMemo, useState } from "react";
import {
  useArticleAffiliatePlacements,
  useSaveArticleAffiliatePlacements,
} from "@/features/affiliate/queries/affiliate.queries";

function getBlockLabel(block, index) {
  const type = block.type
    ? block.type.charAt(0).toUpperCase() + block.type.slice(1)
    : "Content";

  const text = block.text?.trim();

  return {
    title: `${type} ${index + 1}`,
    preview: text || "No text preview available",
  };
}

export default function ArticleAffiliatePlacementEditor({
  articleId,
  blocks = [],
  role,
  articleStatus,
}) {
  const canManage =
    role === "admin" ||
    role === "editor" ||
    (role === "author" && ["draft", "in_review"].includes(articleStatus));

  const placementsQuery = useArticleAffiliatePlacements(articleId, {
    enabled: Boolean(articleId) && canManage,
  });

  const saveMutation = useSaveArticleAffiliatePlacements();

  const [placementOverrides, setPlacementOverrides] = useState(null);

  const data = placementsQuery.data;

  const savedBlocks = useMemo(() => data?.article?.blocks ?? [], [data]);

  const eligibleLinks = useMemo(() => data?.eligibleLinks ?? [], [data]);

  const existingPlacements = useMemo(() => data?.placements ?? [], [data]);

  const initialPlacements = useMemo(
    () =>
      Object.fromEntries(
        existingPlacements.map((placement) => [
          placement.blockId,
          placement.affiliateLinkId,
        ]),
      ),
    [existingPlacements],
  );

  const draftPlacements = placementOverrides ?? initialPlacements;

  const eligibleLinkIds = useMemo(
    () => new Set(eligibleLinks.map((link) => link.id)),
    [eligibleLinks],
  );

  const savedBlockIds = useMemo(
    () =>
      new Set(
        savedBlocks
          .map((block) => block.id)
          .filter((id) => typeof id === "string" && id.length > 0),
      ),
    [savedBlocks],
  );

  const contentStructureChanged = useMemo(() => {
    const currentIds = new Set(
      blocks
        .map((block) => block.id)
        .filter((id) => typeof id === "string" && id.length > 0),
    );

    if (
      blocks.length !== savedBlocks.length ||
      currentIds.size !== blocks.length ||
      savedBlockIds.size !== savedBlocks.length
    ) {
      return true;
    }

    return (
      currentIds.size !== savedBlockIds.size ||
      [...currentIds].some((id) => !savedBlockIds.has(id))
    );
  }, [blocks, savedBlocks, savedBlockIds]);

  const selectedPlacements = useMemo(
    () =>
      Object.entries(draftPlacements)
        .filter(
          ([blockId, affiliateLinkId]) =>
            Boolean(affiliateLinkId) && savedBlockIds.has(blockId),
        )
        .map(([blockId, affiliateLinkId]) => ({
          blockId,
          affiliateLinkId,
        })),
    [draftPlacements, savedBlockIds],
  );

  const hasInvalidSelection = useMemo(
    () =>
      Object.entries(draftPlacements).some(
        ([blockId, affiliateLinkId]) =>
          Boolean(affiliateLinkId) &&
          (!savedBlockIds.has(blockId) ||
            !eligibleLinkIds.has(affiliateLinkId)),
      ),
    [draftPlacements, savedBlockIds, eligibleLinkIds],
  );

  function updatePlacement(blockId, affiliateLinkId) {
    setPlacementOverrides((current) => {
      const next = { ...(current ?? initialPlacements) };

      if (affiliateLinkId) {
        next[blockId] = affiliateLinkId;
      } else {
        delete next[blockId];
      }

      return next;
    });
  }

  function handleSave() {
    saveMutation.mutate({
      articleId,
      placements: selectedPlacements,
    });
  }

  if (!articleId || !canManage) return null;

  return (
    <section className="border border-[#e8e9ed] bg-white p-5 sm:p-7">
      <div className="mb-5">
        <h2 className="text-sm font-bold text-[#14243d]">
          Affiliate link placements
        </h2>

        <p className="mt-1 text-xs leading-5 text-gray-500">
          Associate approved affiliate links with specific content blocks.
          Readers will see the links alongside the relevant article content.
        </p>
      </div>

      {placementsQuery.isLoading ? (
        <p className="text-sm text-gray-500">Loading affiliate placements...</p>
      ) : placementsQuery.isError ? (
        <p className="text-sm text-red-600">
          {placementsQuery.error?.response?.data?.error?.message ||
            "Unable to load affiliate placements."}
        </p>
      ) : savedBlocks.length === 0 ? (
        <p className="text-sm text-gray-500">
          Add and save article content blocks before assigning affiliate links.
        </p>
      ) : (
        <>
          {eligibleLinks.length === 0 && (
            <div className="mb-4 border border-[#eee4d2] bg-[#faf8f3] p-3 text-xs leading-5 text-[#725b35]">
              There are currently no eligible affiliate links. Create or
              activate an affiliate link and its program before assigning it.
            </div>
          )}

          <div className="space-y-4">
            {savedBlocks.map((block, index) => {
              const label = getBlockLabel(block, index);
              const selectedId = draftPlacements[block.id] ?? "";

              const selectedLink = existingPlacements.find(
                (placement) =>
                  placement.blockId === block.id &&
                  placement.affiliateLinkId === selectedId,
              )?.link;

              return (
                <div
                  key={block.id ?? `block-${index}`}
                  className="border border-[#edf0f2] p-4"
                >
                  <div className="mb-3">
                    <p className="text-xs font-semibold text-[#14243d]">
                      {label.title}
                    </p>

                    <p className="mt-1 text-xs leading-5 text-gray-500">
                      {label.preview}
                    </p>
                  </div>

                  <label
                    htmlFor={`affiliate-placement-${block.id ?? index}`}
                    className="mb-1.5 block text-xs font-medium text-[#344054]"
                  >
                    Affiliate link
                  </label>

                  <select
                    id={`affiliate-placement-${block.id ?? index}`}
                    value={selectedId}
                    onChange={(event) =>
                      updatePlacement(block.id, event.target.value)
                    }
                    className="w-full rounded-sm border border-[#dfe2e8] bg-white px-3 py-2.5 text-sm text-[#172033] outline-none focus:border-[#b99658] focus:ring-2 focus:ring-[#b99658]/10"
                  >
                    <option value="">No affiliate link</option>

                    {selectedId && !eligibleLinkIds.has(selectedId) && (
                      <option value={selectedId}>
                        {selectedLink?.title || "Previously assigned link"} (no
                        longer eligible)
                      </option>
                    )}

                    {eligibleLinks.map((link) => (
                      <option key={link.id} value={link.id}>
                        {link.title}
                        {link.program?.name ? ` — ${link.program.name}` : ""}
                      </option>
                    ))}
                  </select>
                </div>
              );
            })}
          </div>

          {contentStructureChanged && (
            <p className="mt-4 border border-[#eee4d2] bg-[#faf8f3] p-3 text-xs leading-5 text-[#725b35]">
              Your article block structure has changed or contains unsaved block
              IDs. Save the article first, then reload this page before saving
              affiliate placements.
            </p>
          )}

          {hasInvalidSelection && (
            <p className="mt-3 text-xs text-red-600">
              Replace or remove affiliate links that are no longer eligible
              before saving.
            </p>
          )}

          {saveMutation.isError && (
            <p className="mt-3 text-xs text-red-600">
              {saveMutation.error?.response?.data?.error?.message ||
                "Unable to save affiliate placements."}
            </p>
          )}

          {saveMutation.isSuccess && (
            <p className="mt-3 text-xs text-green-700">
              Affiliate placements saved successfully.
            </p>
          )}

          <div className="mt-5 flex flex-wrap items-center justify-between gap-3">
            <p className="text-xs text-gray-500">
              {selectedPlacements.length}{" "}
              {selectedPlacements.length === 1 ? "placement" : "placements"}{" "}
              selected
            </p>

            <button
              type="button"
              onClick={handleSave}
              disabled={
                saveMutation.isPending ||
                contentStructureChanged ||
                hasInvalidSelection
              }
              className="rounded-sm bg-[#14243d] px-5 py-2.5 text-xs font-semibold text-white transition hover:bg-[#203653] disabled:cursor-not-allowed disabled:opacity-50"
            >
              {saveMutation.isPending
                ? "Saving placements..."
                : "Save placements"}
            </button>
          </div>
        </>
      )}
    </section>
  );
}

function handleDatabaseError(error) {
  if (error.code === "23505") {
    throw new AuthError(
      "An article with this slug already exists. Choose a different slug.",
      409,
      "ARTICLE_SLUG_EXISTS",
    );
  }

  if (error.code === "23503") {
    throw new AuthError(
      "The selected author or category is no longer available.",
      409,
      "ARTICLE_REFERENCE_INVALID",
    );
  }

  if (error.code === "23514" || error.code === "P0001") {
    throw new AuthError(
      error.message || "The article violates an editorial publishing rule.",
      422,
      "ARTICLE_RULE_VIOLATION",
    );
  }

  if (error.code === "42501") {
    throw new AuthError(
      "You do not have permission to perform this operation.",
      403,
      "ARTICLE_PERMISSION_DENIED",
    );
  }

  if (error.code === "23502") {
    throw new AuthError(
      "A required article field is missing.",
      422,
      "ARTICLE_REQUIRED_FIELD_MISSING",
    );
  }

  throw error;
}

export async function GET(request) {
  try {
    const { supabase, user, profile } = await requireRole([
      "author",
      "editor",
      "admin",
    ]);

    const url = new URL(request.url);

    const rawFilters = Object.fromEntries(url.searchParams.entries());
    const filterResult = articleListFiltersSchema.safeParse(rawFilters);

    if (!filterResult.success) {
      throw filterResult.error;
    }

    const filters = filterResult.data;
    const isAuthor = profile.role === "author";

    let query = supabase
      .from("articles")
      .select(articleFields, { count: "exact" })
      .order(filters.sort, {
        ascending: filters.order === "asc",
        nullsFirst: false,
      });

    if (isAuthor) {
      query = query.eq("author_id", user.id);
    } else if (filters.authorId) {
      query = query.eq("author_id", filters.authorId);
    }

    if (filters.status) {
      query = query.eq("status", filters.status);
    }

    if (filters.categoryId) {
      query = query.eq("category_id", filters.categoryId);
    }

    if (filters.search) {
      query = query.ilike("title", `%${filters.search}%`);
    }

    const from = (filters.page - 1) * filters.pageSize;
    const to = from + filters.pageSize - 1;

    const { data, error, count } = await query.range(from, to);

    if (error) throw error;

    const total = count ?? 0;

    return successResponse({
      items: data ?? [],
      pagination: {
        page: filters.page,
        pageSize: filters.pageSize,
        total,
        totalPages: Math.ceil(total / filters.pageSize),
      },
    });
  } catch (error) {
    return handleApiError(error);
  }
}

export async function POST(request) {
  try {
    assertSameOrigin(request);

    const { supabase, user, profile } = await requireRole([
      "author",
      "editor",
      "admin",
    ]);

    const body = await readValidatedBody(request, createArticleSchema);

    const isAuthor = profile.role === "author";

    if (isAuthor && body.authorId && body.authorId !== user.id) {
      throw new AuthError(
        "Authors can only create articles under their own profile.",
        403,
        "ARTICLE_AUTHOR_ASSIGNMENT_FORBIDDEN",
      );
    }

    if (isAuthor && !["draft", "in_review"].includes(body.status)) {
      throw new AuthError(
        "Authors can only create draft or review-submission articles.",
        403,
        "ARTICLE_STATUS_FORBIDDEN",
      );
    }

    const authorId = isAuthor ? user.id : body.authorId || user.id;
    const slug = body.slug || slugify(body.title);

    if (!slug) {
      throw new AuthError(
        "A valid article slug could not be generated from the title.",
        422,
        "INVALID_ARTICLE_SLUG",
      );
    }

    const { data, error } = await supabase
      .from("articles")
      .insert({
        title: body.title,
        slug,
        excerpt: body.excerpt ?? null,
        content: body.content,
        category_id: body.categoryId,
        author_id: authorId,
        status: body.status,
        featured_image_path: body.featuredImagePath ?? null,
        seo_title: body.seoTitle ?? null,
        seo_description: body.seoDescription ?? null,
        focus_keyword: body.focusKeyword ?? null,
        canonical_url: body.canonicalUrl ?? null,
        no_index: body.noIndex,
        scheduled_at: body.scheduledAt ?? null,
      })
      .select(articleFields)
      .single();

    if (error) handleDatabaseError(error);

    if (data.status === "in_review" && profile.role === "author") {
      await createNotificationsForRolesSafely({
        roles: ["editor", "admin"],
        actorId: user.id,
        type: "moderation",
        title: "Article submitted for review",
        message: `"${data.title}" is ready for editorial review.`,
        actionUrl: "/dashboard/articles",
        metadata: {
          event: "article_submitted_for_review",
          articleId: data.id,
          articleTitle: data.title,
        },
        dedupeKey: `article:${data.id}:submitted-for-review:${data.updated_at}`,
      });
    }

    if (
      ["editor", "admin"].includes(profile.role) &&
      ["published", "scheduled"].includes(data.status) &&
      data.author_id !== user.id
    ) {
      const isPublished = data.status === "published";

      await createNotificationSafely({
        userId: data.author_id,
        actorId: user.id,
        type: "article",
        title: isPublished
          ? "Your article has been published"
          : "Your article has been scheduled",
        message: isPublished
          ? `"${data.title}" has been published.`
          : `"${data.title}" has been scheduled for publication.`,
        actionUrl: "/dashboard/articles",
        metadata: {
          event: isPublished ? "article_published" : "article_scheduled",
          articleId: data.id,
          articleTitle: data.title,
          status: data.status,
          publishedAt: data.published_at,
          scheduledAt: data.scheduled_at,
        },
        dedupeKey: `article:${data.id}:${data.status}:${data.updated_at}`,
      });
    }

    return successResponse(data, 201, "Article created successfully.");
  } catch (error) {
    return handleApiError(error);
  }
}

async function getArticle(supabase, articleId) {
  const { data, error } = await supabase
    .from("articles")
    .select(articleFields)
    .eq("id", articleId)
    .maybeSingle();

  if (error) throw error;

  if (!data) {
    throw new AuthError(
      "The requested article was not found.",
      404,
      "ARTICLE_NOT_FOUND",
    );
  }

  return data;
}

export async function PATCH(request, { params }) {
  try {
    assertSameOrigin(request);

    const { supabase, user, profile } = await requireRole([
      "author",
      "editor",
      "admin",
    ]);

    const { articleId } = await params;
    const idResult = articleIdSchema.safeParse(articleId);

    if (!idResult.success) {
      throw new AuthError(
        "The article identifier is invalid.",
        400,
        "INVALID_ARTICLE_ID",
      );
    }

    const body = await readValidatedBody(request, updateArticleSchema);
    const article = await getArticle(supabase, articleId);

    const isAuthor = profile.role === "author";

    if (isAuthor) {
      if (!["draft", "in_review"].includes(article.status)) {
        throw new AuthError(
          "You can only edit articles that are drafts or awaiting review.",
          403,
          "ARTICLE_STATUS_NOT_EDITABLE",
        );
      }
      if (article.author_id !== user.id) {
        throw new AuthError(
          "You can only update your own articles.",
          403,
          "ARTICLE_OWNERSHIP_REQUIRED",
        );
      }

      if (body.authorId !== undefined && body.authorId !== user.id) {
        throw new AuthError(
          "Authors cannot reassign an article to another author.",
          403,
          "ARTICLE_AUTHOR_ASSIGNMENT_FORBIDDEN",
        );
      }

      if (
        body.status !== undefined &&
        !["draft", "in_review"].includes(body.status)
      ) {
        throw new AuthError(
          "Authors can only save articles as drafts or submit them for review.",
          403,
          "ARTICLE_STATUS_FORBIDDEN",
        );
      }
    }

    const updates = {};

    const fieldMap = {
      title: "title",
      slug: "slug",
      excerpt: "excerpt",
      content: "content",
      categoryId: "category_id",
      authorId: "author_id",
      status: "status",
      featuredImagePath: "featured_image_path",
      seoTitle: "seo_title",
      seoDescription: "seo_description",
      focusKeyword: "focus_keyword",
      canonicalUrl: "canonical_url",
      noIndex: "no_index",
      scheduledAt: "scheduled_at",
    };

    for (const [inputField, databaseField] of Object.entries(fieldMap)) {
      if (Object.hasOwn(body, inputField)) {
        updates[databaseField] = body[inputField];
      }
    }

    if (
      !isAuthor &&
      (body.status === "scheduled" ||
        (body.status === undefined && article.status === "scheduled"))
    ) {
      const scheduledAt =
        body.scheduledAt !== undefined
          ? body.scheduledAt
          : article.scheduled_at;

      if (!scheduledAt || new Date(scheduledAt).getTime() <= Date.now()) {
        throw new AuthError(
          "A scheduled article must have a future publication date and time.",
          422,
          "INVALID_SCHEDULED_DATE",
        );
      }
    }

    const { data, error } = await supabase
      .from("articles")
      .update(updates)
      .eq("id", articleId)
      .select(articleFields)
      .single();

    if (error) handleDatabaseError(error);

    // Only notify when the article's status actually changes.
    if (article.status !== data.status) {
      // Author submitted an article for editorial review.
      if (profile.role === "author" && data.status === "in_review") {
        await createNotificationsForRolesSafely({
          roles: ["editor", "admin"],
          actorId: user.id,
          type: "moderation",
          title: "Article submitted for review",
          message: `"${data.title}" is ready for editorial review.`,
          actionUrl: "/dashboard/articles",
          metadata: {
            event: "article_submitted_for_review",
            articleId: data.id,
            articleTitle: data.title,
          },
          dedupeKey: `article:${data.id}:submitted-for-review:${data.updated_at}`,
        });
      }

      // Editor or admin published or scheduled an article.
      if (
        ["editor", "admin"].includes(profile.role) &&
        ["published", "scheduled"].includes(data.status) &&
        data.author_id !== user.id
      ) {
        const isPublished = data.status === "published";

        await createNotificationSafely({
          userId: data.author_id,
          actorId: user.id,
          type: "article",
          title: isPublished
            ? "Your article has been published"
            : "Your article has been scheduled",
          message: isPublished
            ? `"${data.title}" has been published.`
            : `"${data.title}" has been scheduled for publication.`,
          actionUrl: "/dashboard/articles",
          metadata: {
            event: isPublished ? "article_published" : "article_scheduled",
            articleId: data.id,
            articleTitle: data.title,
            status: data.status,
            publishedAt: data.published_at,
            scheduledAt: data.scheduled_at,
          },
          dedupeKey: `article:${data.id}:${data.status}:${data.updated_at}`,
        });
      }
    }

    return successResponse(data, 200, "Article updated successfully.");
  } catch (error) {
    return handleApiError(error);
  }
}

export async function DELETE(request, { params }) {
  try {
    assertSameOrigin(request);

    const { supabase } = await requireRole(["editor", "admin"]);

    const { articleId } = await params;
    const idResult = articleIdSchema.safeParse(articleId);

    if (!idResult.success) {
      throw new AuthError(
        "The article identifier is invalid.",
        400,
        "INVALID_ARTICLE_ID",
      );
    }

    await getArticle(supabase, articleId);

    const { error } = await supabase
      .from("articles")
      .delete()
      .eq("id", articleId);

    if (error) handleDatabaseError(error);

    return successResponse(null, 200, "Article deleted successfully.");
  } catch (error) {
    return handleApiError(error);
  }
}

  if (
    bytes.length >= 3 &&
    bytes[0] === 0xff &&
    bytes[1] === 0xd8 &&
    bytes[2] === 0xff
  ) {
    return allowedMimeTypes.jpeg;
  }

  if (
    bytes.length >= 8 &&
    bytes[0] === 0x89 &&
    bytes[1] === 0x50 &&
    bytes[2] === 0x4e &&
    bytes[3] === 0x47 &&
    bytes[4] === 0x0d &&
    bytes[5] === 0x0a &&
    bytes[6] === 0x1a &&
    bytes[7] === 0x0a
  ) {
    return allowedMimeTypes.png;
  }

  if (
    bytes.length >= 12 &&
    String.fromCharCode(...bytes.slice(0, 4)) === "RIFF" &&
    String.fromCharCode(...bytes.slice(8, 12)) === "WEBP"
  ) {
    return allowedMimeTypes.webp;
  }

  if (
    bytes.length >= 12 &&
    String.fromCharCode(...bytes.slice(4, 8)) === "ftyp" &&
    ["avif", "avis"].includes(String.fromCharCode(...bytes.slice(8, 12)))
  ) {
    return allowedMimeTypes.avif;
  }

  return null;
}

function getExtension(mimeType) {
  const extensions = {
    "image/jpeg": "jpg",
    "image/png": "png",
    "image/webp": "webp",
    "image/avif": "avif",
  };

  return extensions[mimeType];
}

function sanitizeFilename(filename) {
  const sanitized = String(filename || "image")
    .replace(/[\\/\u0000-\u001f\u007f]/g, "_")
    .trim()
    .slice(0, 255);

  return sanitized || "image";
}

function mapDatabaseError(error) {
  if (error.code === "23505") {
    throw new AuthError(
      "A media record with this storage path already exists.",
      409,
      "MEDIA_ALREADY_EXISTS",
    );
  }

  if (error.code === "23514" || error.code === "23503") {
    throw new AuthError(
      "The uploaded media did not satisfy the storage rules.",
      422,
      "MEDIA_VALIDATION_FAILED",
    );
  }

  if (error.code === "42501") {
    throw new AuthError(
      "You do not have permission to manage this media.",
      403,
      "MEDIA_PERMISSION_DENIED",
    );
  }

  throw error;
}

async function getPreviewUrl(supabase, asset) {
  if (asset.status === "archived") {
    return null;
  }

  if (asset.bucket_id === "trendpilot-media") {
    const { data } = supabase.storage
      .from("trendpilot-media")
      .getPublicUrl(asset.object_path);

    return data?.publicUrl ?? null;
  }

  const { data, error } = await supabase.storage
    .from(BUCKET)
    .createSignedUrl(asset.object_path, 3600);

  if (error) {
    return null;
  }

  return data?.signedUrl ?? null;
}

export async function GET(request) {
  try {
    const { supabase } = await requireRole(["author", "editor", "admin"]);

    const url = new URL(request.url);

    const filterResult = mediaListFiltersSchema.safeParse(
      Object.fromEntries(url.searchParams.entries()),
    );

    if (!filterResult.success) {
      throw filterResult.error;
    }

    const filters = filterResult.data;

    let query = supabase
      .from("media_assets")
      .select(mediaFields, { count: "exact" })
      .order("created_at", { ascending: false });

    if (filters.status) {
      query = query.eq("status", filters.status);
    }

    if (filters.purpose === "unclassified") {
      query = query.is("purpose", null);
    } else if (filters.purpose) {
      query = query.eq("purpose", filters.purpose);
    }

    const from = (filters.page - 1) * filters.pageSize;
    const to = from + filters.pageSize - 1;

    const { data, error, count } = await query.range(from, to);

    if (error) {
      throw error;
    }

    const items = await Promise.all(
      (data ?? []).map(async (asset) => ({
        ...asset,
        preview_url: await getPreviewUrl(supabase, asset),
      })),
    );

    const total = count ?? 0;

    return successResponse({
      items,
      pagination: {
        page: filters.page,
        pageSize: filters.pageSize,
        total,
        totalPages: Math.ceil(total / filters.pageSize),
      },
    });
  } catch (error) {
    return handleApiError(error);
  }
}

export async function POST(request) {
  let uploadedObjectPath = null;

  try {
    assertSameOrigin(request);

    const { supabase, user } = await requireRole(["author", "editor", "admin"]);

    const formData = await request.formData();
    const file = formData.get("file");

    if (
      !file ||
      typeof file.arrayBuffer !== "function" ||
      typeof file.size !== "number"
    ) {
      throw new AuthError(
        "Select an image to upload.",
        400,
        "MEDIA_FILE_REQUIRED",
      );
    }

    if (file.size <= 0 || file.size > MAX_FILE_SIZE) {
      throw new AuthError(
        "Images must be larger than zero bytes and no larger than 10 MB.",
        413,
        "MEDIA_FILE_SIZE_INVALID",
      );
    }

    const metadataResult = mediaMetadataSchema.safeParse({
      purpose:
        typeof formData.get("purpose") === "string"
          ? formData.get("purpose")
          : undefined,
      altText:
        typeof formData.get("altText") === "string"
          ? formData.get("altText")
          : "",
      caption:
        typeof formData.get("caption") === "string"
          ? formData.get("caption")
          : "",
    });

    if (!metadataResult.success) {
      throw metadataResult.error;
    }

    const metadata = metadataResult.data;

    const header = new Uint8Array(await file.slice(0, 16).arrayBuffer());

    const detectedMime = detectImageMime(header);

    if (!detectedMime || (file.type && file.type !== detectedMime)) {
      throw new AuthError(
        "The selected file is not a supported, valid image.",
        415,
        "MEDIA_TYPE_INVALID",
      );
    }

    const extension = getExtension(detectedMime);
    const originalFilename = sanitizeFilename(file.name);

    const now = new Date();
    const year = now.getUTCFullYear();
    const month = String(now.getUTCMonth() + 1).padStart(2, "0");

    uploadedObjectPath = `${user.id}/${year}/${month}/${randomUUID()}.${extension}`;

    const { error: uploadError } = await supabase.storage
      .from(BUCKET)
      .upload(uploadedObjectPath, file, {
        cacheControl: "3600",
        contentType: detectedMime,
        upsert: false,
      });

    if (uploadError) {
      throw new AuthError(
        "The image could not be uploaded to secure storage.",
        502,
        "MEDIA_STORAGE_UPLOAD_FAILED",
      );
    }

    const { data: asset, error: insertError } = await supabase
      .from("media_assets")
      .insert({
        owner_id: user.id,
        created_by: user.id,
        bucket_id: BUCKET,
        object_path: uploadedObjectPath,
        original_filename: originalFilename,
        mime_type: detectedMime,
        file_size_bytes: file.size,
        purpose: metadata.purpose,
        alt_text: metadata.altText,
        caption: metadata.caption,
        status: "staged",
      })
      .select(mediaFields)
      .single();

    if (insertError) {
      mapDatabaseError(insertError);
    }

    const previewUrl = await getPreviewUrl(supabase, asset);

    return successResponse(
      {
        ...asset,
        preview_url: previewUrl,
      },
      201,
      "Image uploaded successfully and is awaiting editorial approval.",
    );
  } catch (error) {
    if (uploadedObjectPath) {
      try {
        const adminSupabase = createSupabaseAdminClient();

        await adminSupabase.storage.from(BUCKET).remove([uploadedObjectPath]);
      } catch {
        // Preserve the original upload or database error.
      }
    }

    return handleApiError(error);
  }
}

export function getArticleReadingTime(content) {
  const blocks = Array.isArray(content?.content) ? content.content : [];

  const words = blocks.reduce((total, block) => {
    const text =
      typeof block.text === "string"
        ? block.text
        : Array.isArray(block.items)
          ? block.items.join(" ")
          : "";

    return total + (text.trim() ? text.trim().split(/\s+/).length : 0);
  }, 0);

  return Math.max(1, Math.ceil(words / 200));
}

export function getPublicImageUrl(supabase, imagePath) {
  if (!imagePath) return null;

  const { data } = supabase.storage
    .from("trendpilot-media")
    .getPublicUrl(imagePath);

  return data?.publicUrl || null;
}

export function getArticleExcerpt(article) {
  if (article?.excerpt?.trim()) return article.excerpt.trim();

  const blocks = Array.isArray(article?.content?.content)
    ? article.content.content
    : [];

  const text = blocks
    .map((block) => {
      if (typeof block.text === "string") return block.text;
      if (Array.isArray(block.items)) return block.items.join(" ");
      return "";
    })
    .filter(Boolean)
    .join(" ")
    .replace(/\s+/g, " ")
    .trim();

  if (text.length <= 180) return text;

  return `${text.slice(0, 177).trimEnd()}...`;
}

export async function getPublishedCategoryPath(slugs) {
  if (!Array.isArray(slugs) || slugs.length === 0) return null;

  const categories = await getPublishedCategories();
  let parentId = null;
  let selected = null;

  for (let index = 0; index < slugs.length; index += 1) {
    const slug = slugs[index];

    selected = categories.find(
      (category) => category.slug === slug && category.parent_id === parentId,
    );

    if (!selected) return null;

    parentId = selected.id;
  }

  return selected;
}

export function getCategoryAncestors(category, categories) {
  const ancestors = [];
  let current = category;

  while (current) {
    ancestors.unshift(current);

    if (!current.parent_id) break;

    current = categories.find((item) => item.id === current.parent_id);

    if (!current) return [];
  }

  return ancestors;
}

export function getCategoryPathname(slugs) {
  if (!Array.isArray(slugs) || slugs.length === 0) {
    return "/articles";
  }

  return `/category/${slugs.map((slug) => encodeURIComponent(slug)).join("/")}`;
}

export function getCategoryUrl(category, categories) {
  const ancestors = getCategoryAncestors(category, categories);

  if (!ancestors.length) return "/articles";

  return getCategoryPathname(ancestors.map((item) => item.slug));
}

export function getCategoryDescendantMicroIds(category, categories) {
  const descendants = new Set([category.id]);

  let changed = true;

  while (changed) {
    changed = false;

    for (const item of categories) {
      if (item.parent_id && descendants.has(item.parent_id)) {
        if (!descendants.has(item.id)) {
          descendants.add(item.id);
          changed = true;
        }
      }
    }
  }

  return categories
    .filter((item) => item.level === "micro" && descendants.has(item.id))
    .map((item) => item.id);
}

export async function getPublishedArticles({ limit = 12, categoryIds } = {}) {
  await connection();
  const supabase = await createSupabaseServerClient();

  let query = supabase
    .from("articles")
    .select(ARTICLE_FIELDS)
    .eq("status", "published")
    .not("published_at", "is", null)
    .lte("published_at", new Date().toISOString())
    .order("published_at", { ascending: false })
    .limit(limit);

  if (Array.isArray(categoryIds)) {
    if (categoryIds.length === 0) return [];

    query = query.in("category_id", categoryIds);
  }

  const { data, error } = await query;

  if (error) {
    console.error("Public article query failed:", error);
    return [];
  }

  const articles = data || [];
  if (!articles.length) return [];

  const categories = await getPublishedCategories();

  const categoryMap = new Map(
    categories.map((category) => [category.id, category]),
  );

  return articles.map((article) => {
    const category = categoryMap.get(article.category_id) || null;

    return {
      ...article,
      category,
      categoryUrl: category
        ? getCategoryUrl(category, categories)
        : "/articles",
      imageUrl: getPublicImageUrl(supabase, article.featured_image_path),
      excerptText: getArticleExcerpt(article),
      readingTime: getArticleReadingTime(article.content),
    };
  });
}

export async function getPublishedArticlesPage({
  page = 1,
  pageSize = 12,
  categoryIds = [],
} = {}) {
  await connection();

  const supabase = await createSupabaseServerClient();

  const safePage = Math.max(1, Math.floor(Number(page) || 1));
  const safePageSize = Math.min(
    24,
    Math.max(1, Math.floor(Number(pageSize) || 12)),
  );

  if (!Array.isArray(categoryIds) || categoryIds.length === 0) {
    return {
      articles: [],
      total: 0,
      page: safePage,
      pageSize: safePageSize,
      totalPages: 1,
    };
  }

  const from = (safePage - 1) * safePageSize;
  const to = from + safePageSize - 1;

  const { data, error, count } = await supabase
    .from("articles")
    .select(ARTICLE_FIELDS, { count: "exact" })
    .eq("status", "published")
    .not("published_at", "is", null)
    .lte("published_at", new Date().toISOString())
    .in("category_id", categoryIds)
    .order("published_at", { ascending: false })
    .range(from, to);

  if (error) {
    console.error("Paginated public article query failed:", error);

    return {
      articles: [],
      total: 0,
      page: safePage,
      pageSize: safePageSize,
      totalPages: 1,
    };
  }

  const articles = data || [];
  const total = count || 0;

  if (articles.length === 0) {
    return {
      articles: [],
      total,
      page: safePage,
      pageSize: safePageSize,
      totalPages: Math.max(1, Math.ceil(total / safePageSize)),
    };
  }

  const categories = await getPublishedCategories();

  const categoryMap = new Map(
    categories.map((category) => [category.id, category]),
  );

  const enrichedArticles = articles.map((article) => {
    const category = categoryMap.get(article.category_id) || null;

    return {
      ...article,
      category,
      categoryUrl: category
        ? getCategoryUrl(category, categories)
        : "/articles",
      imageUrl: getPublicImageUrl(supabase, article.featured_image_path),
      excerptText: getArticleExcerpt(article),
      readingTime: getArticleReadingTime(article.content),
    };
  });

  return {
    articles: enrichedArticles,
    total,
    page: safePage,
    pageSize: safePageSize,
    totalPages: Math.max(1, Math.ceil(total / safePageSize)),
  };
}

export async function searchPublishedArticles({ query, limit = 24 } = {}) {
  await connection();

  const searchTerm = typeof query === "string" ? query.trim() : "";

  if (!searchTerm || searchTerm.length < 2) {
    return [];
  }

  const supabase = await createSupabaseServerClient();

  const safeLimit = Math.min(48, Math.max(1, Math.floor(Number(limit) || 24)));

  // Escape SQL LIKE wildcard characters so they are treated literally.
  const pattern = `%${searchTerm.replace(/[\\%_]/g, "\\$&")}%`;

  const searchableFields = [
    "title",
    "excerpt",
    "seo_title",
    "seo_description",
    "focus_keyword",
  ];

  const results = await Promise.all(
    searchableFields.map((field) =>
      supabase
        .from("articles")
        .select(ARTICLE_FIELDS)
        .eq("status", "published")
        .not("published_at", "is", null)
        .lte("published_at", new Date().toISOString())
        .ilike(field, pattern)
        .order("published_at", { ascending: false })
        .limit(safeLimit),
    ),
  );

  const articlesById = new Map();

  for (const result of results) {
    if (result.error) {
      console.error("Public article search failed:", result.error);
      continue;
    }

    for (const article of result.data || []) {
      articlesById.set(article.id, article);
    }
  }

  const articles = [...articlesById.values()]
    .sort(
      (a, b) =>
        new Date(b.published_at).getTime() - new Date(a.published_at).getTime(),
    )
    .slice(0, safeLimit);

  if (!articles.length) {
    return [];
  }

  const categories = await getPublishedCategories();

  const categoryMap = new Map(
    categories.map((category) => [category.id, category]),
  );

  return articles
    .map((article) => {
      const category = categoryMap.get(article.category_id) || null;

      // Do not expose articles whose category is not publicly available.
      if (!category) return null;

      return {
        ...article,
        category,
        categoryUrl: getCategoryUrl(category, categories),
        imageUrl: getPublicImageUrl(supabase, article.featured_image_path),
        excerptText: getArticleExcerpt(article),
        readingTime: getArticleReadingTime(article.content),
      };
    })
    .filter(Boolean);
}
